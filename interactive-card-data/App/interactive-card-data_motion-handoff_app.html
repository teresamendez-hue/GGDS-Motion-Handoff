Implementación Dart (Flutter)

final motion = context.motionTokens;
final spring = motion.springSm; // motion-spring-sm (mass: 1.0, stiffness: 400, damping: 35)
final disableAnimations = MediaQuery.of(context).disableAnimations;

double scale = 1.0;
double focusRingOpacity = 0.0;

void onTapDown() {
  if (disableAnimations) {
    scale = 0.98;
    return;
  }
  // animate scale 1.0 -> 0.98 with motion.springSm
}

void onTapUp() {
  if (disableAnimations) {
    scale = 1.0;
    return;
  }
  // animate scale 0.98 -> 1.0 with motion.springSm
}

void onFocusChange(bool focused) {
  if (disableAnimations) {
    focusRingOpacity = focused ? 1.0 : 0.0;
    return;
  }
  // animate focusRingOpacity 0 <-> 1 with motion.springSm
}
